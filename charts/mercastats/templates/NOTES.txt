1. Acceso a la web:
{{- if $baseUrl }}

     {{ $baseUrl }}
{{- else if .Values.httpRoute.enabled }}

     HTTPRoute: {{ join ", " (.Values.httpRoute.hostnames | default (list "(sin hostnames)")) }}
     (pon config.baseUrl con la URL pública para Gmail OAuth y COOKIE_SECURE automático)
{{- else if and .Values.service.enabled (eq .Values.service.type "NodePort") }}

     export NODE_PORT=$(kubectl get svc {{ $fullname }} -n {{ $ns }} -o jsonpath="{.spec.ports[0].nodePort}")
     export NODE_IP=$(kubectl get nodes -o jsonpath="{.items[0].status.addresses[0].address}")
     echo "http://$NODE_IP:$NODE_PORT"
{{- else if and .Values.service.enabled (eq .Values.service.type "LoadBalancer") }}

     La IP externa puede tardar unos minutos (kubectl get svc -w {{ $fullname }} -n {{ $ns }}):
     export SERVICE_IP=$(kubectl get svc {{ $fullname }} -n {{ $ns }} -o jsonpath="{.status.loadBalancer.ingress[0]['ip','hostname']}")
     echo "http://$SERVICE_IP:{{ .Values.service.port }}"
{{- else }}

     kubectl port-forward -n {{ $ns }} svc/{{ $fullname }} 8080:{{ .Values.service.port }}
     y abre http://127.0.0.1:8080
{{- end }}
